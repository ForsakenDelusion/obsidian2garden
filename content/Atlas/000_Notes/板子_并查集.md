```cpp
const int N = 200005; // 根据题目要求开大小
int p[N]; // parent 数组，p[i] 表示 i 的父节点

// 1. 初始化 (Initial)
// 这一步千万别忘了！一开始每个人都是自己的老大
void init_dsu(int n) {
    for (int i = 0; i <= n; i++) p[i] = i; // 0-based 或 1-based 通杀
}

// 2. 查 (Find) + 路径压缩 (Path Compression)
// 这是并查集的灵魂，加了路径压缩后，复杂度接近 O(1)
int find(int x) {
    if (p[x] != x) p[x] = find(p[x]); // 递归找老大，并顺手把沿途所有人直接挂到老大下面
    return p[x];
}

// 3. 并 (Union)
// 也就是把两个集合合并，返回 bool 表示是否成功合并（原本不在同一集合）
bool unite(int x, int y) {
    int rootX = find(x);
    int rootY = find(y);
    
    if (rootX != rootY) {
        p[rootX] = rootY; // 把 x 的老大归顺给 y
        return true;      // 合并成功
    }
    return false;         // 已经是同一家了，无需合并
}
```
